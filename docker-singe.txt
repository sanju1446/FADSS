
========================================
PART 1 - DOCKER CLI (REDIS)
========================================

docker --version
docker pull redis
docker run --name my-redis -d redis
docker ps
docker exec -it my-redis redis-cli

SET name "Alice"
GET name
exit

docker stop my-redis
docker start my-redis
docker stop my-redis
docker rm my-redis
docker rmi redis
docker ps -a
docker images
